scripted_map_modes = {
	culture_map_mode = {
		top = { type = none }
		bottom = {
			type = state
			# FROM = the state being drawn. One array lookup per state instead of a ~390-branch else_if chain.
			color = {
				set_temp_variable = { vod_n = 0 }
				if = {
					limit = { FROM = { has_variable = nationality } }
					set_temp_variable = { vod_n = FROM.nationality }
				}
				# ids outside the table fall back to index 0
				if = {
					limit = { check_variable = { vod_n > 392 } }
					set_temp_variable = { vod_n = 0 }
				}
				set_temp_variable = { red = global.culture_r^vod_n }
				set_temp_variable = { green = global.culture_g^vod_n }
				set_temp_variable = { blue = global.culture_b^vod_n }
				set_temp_variable = { alpha = 0.95 }
			}
		}
		far_text = none
		near_text = state
		update_daily = no
	}
}
